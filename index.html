const robot = document.getElementById("robot"); const nameDisplay = document.getElementById("name"); const message = document.getElementById("message"); const skins = { thinking: { name: "Pensando", message: "Emmet está pensando..." }, confusion: { name: "Confusión", message: "Emmet está confundido..." }, astronomy: { name: "Astronomía", message: "¡Explorando el universo!" }, epic: { name: "Épico", message: "¡Emmet está en modo épico!" }, energy: { name: "Energía", message: "¡Energía al máximo!" } }; const skinOrder = [ "thinking", "confusion", "astronomy", "epic", "energy" ]; let currentSkin = 0; function changeSkin(skin) { if (!skins[skin]) return; currentSkin = skinOrder.indexOf(skin); robot.className = ""; robot.classList.add("skin-" + skin); nameDisplay.textContent = skins[skin].name; message.textContent = skins[skin].message; } function nextSkin() { currentSkin++; if (currentSkin >= skinOrder.length) { currentSkin = 0; } changeSkin(skinOrder[currentSkin]); }
